Account Statement | Date: ${date}

Karigar: ${k.name}

`; const rows = Math.max(k.dr.length, k.cr.length); for (let i = 0; i < rows; i++) { pdfHTML += ` `; } pdfHTML += `

${k.dr[i]?.date || '-'} ${k.dr[i]?.wt.toFixed(3) || '-'} ${k.cr[i]?.date || '-'} ${k.cr[i]?.wt.toFixed(3) || '-'} ${k.cr[i]?.rounded.toFixed(1) || '-'}

Total Jama: ${document.getElementById('dispCr').innerText} g

Balance Gold: ${document.getElementById('dispBal').innerText} g

Settlement Amount: ${document.getElementById('dispRs').innerText}

`; html2pdf().from(pdfHTML).set({ margin: 10, filename: `Gopi_${k.name}_${date}.pdf` }).save(); }
